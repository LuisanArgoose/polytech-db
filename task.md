-- Table: StudentGroup
CREATE TABLE "StudentGroup" (
    "GroupCode" VARCHAR(20) PRIMARY KEY,
    "CourseYear" SMALLINT NOT NULL,
    "StudentCount" INTEGER NOT NULL CHECK ("StudentCount" >= 0),
    "Faculty" VARCHAR(100) NOT NULL
);

-- Table: Subject
CREATE TABLE "Subject" (
    "SubjectName" VARCHAR(150) PRIMARY KEY,
    "Department" VARCHAR(100) NOT NULL,
    "TotalHours" INTEGER NOT NULL CHECK ("TotalHours" > 0)
);

-- Table: ExamSession (Associative Entity)
CREATE TABLE "ExamSession" (
    "GroupCode" VARCHAR(20) NOT NULL,
    "SubjectName" VARCHAR(150) NOT NULL,
    "ExamDate" DATE NOT NULL,
    "TeacherFullName" VARCHAR(150) NOT NULL,
    "AssessmentType" VARCHAR(50) NOT NULL, -- e.g., 'Exam', 'Pass/Fail'
    
    -- Composite Primary Key
    PRIMARY KEY ("GroupCode", "SubjectName", "ExamDate"),
    
    -- Foreign Keys
    CONSTRAINT "FK_ExamSession_StudentGroup" 
        FOREIGN KEY ("GroupCode") REFERENCES "StudentGroup" ("GroupCode") 
        ON UPDATE CASCADE ON DELETE CASCADE,
        
    CONSTRAINT "FK_ExamSession_Subject" 
        FOREIGN KEY ("SubjectName") REFERENCES "Subject" ("SubjectName") 
        ON UPDATE CASCADE ON DELETE CASCADE
);

Практическое задание №1.4 Контроль целостности данных
Постановка задачи
Практическое задание посвящено контролю целостности данных, который
производится с помощью механизма транзакций и триггеров. Транзакции позволяют
рассматривать группу операций как единое целое, либо отрабатывают все операции,
либо ни одной. Это позволяет избегать несогласованности данных. Триггеры
позволяют проверять целостность данных в момент выполнения транзакций,
поддерживать целостность, внося изменения, и откатывать транзакции, приводящие к
потере целостности.
Необходимо подготовить SQL-скрипты для проверки наличия аномалий (потерянных
изменений, грязных чтений, неповторяющихся чтений, фантомов) при параллельном
исполнении транзакций на различных уровнях изолированности SQL/92 (READ
UNCOMMITTED, READ COMMITTED, REPEATABLE READ, SERIALIZABLE).
Подготовленные скрипты должны работать с одной из таблиц, созданных в
практическом задании №1.2. Для проверки наличия аномалий потребуются два
параллельных сеанса, операторы в которых выполняются пошагово:
• Установить в обоих сеансах уровень изоляции READ COMMITED/READ
UNCOMMITTED (для postgresql это синонимы). Выполнить сценарии проверки
наличия аномалий потерянных изменений, грязных чтений, неповторяющихся
чтений, фантомов.
• Установить в обоих сеансах уровень изоляции REPEATABLE READ. Выполнить
сценарии проверки наличия аномалий неповторяющихся чтений и фантомов
(проверить те же скрипты, что и для READ COMMITED), сериализации.
• Установить в обоих сеансах уровень изоляции SERIALIZABLE. Выполнить
сценарий проверки фиксации аномалии сериализации (проверить тот же
скрипт, что и для REPEATABLE READ).
Необходимо составить скрипт для создания 1-2х триггеров, а также подготовить
несколько запросов для проверки и демонстрации его полезных свойств:
• Изменение данных для сохранения целостности.
• Проверка транзакций и их откат в случае нарушения целостности.

жесткое изменение значений в 2-транзакциях (set Filed=Value1 и set Filed=Value2), а
более мягкую трактовку по инкрементам (set Filed=Filed+Value1 и set Filed=Filed+Value2).
Т.е. при отсутствии аномалии сохраняются оба инкремента по двум транзакциям.
- Проверка аномалии сериализации по документации PostgreSQL означает, что результат
параллельно выполняемых сериализуемых транзакций может не согласовываться с
результатом этих же транзакций, выполняемых по очереди. Это значит, что для
проверки аномалии нужно получить 2 состояния БД (например, select *): одно после
параллельного выполнения и другое после последовательного выполнения транзакций
(каждое выполнение начинается с одного и того же исходного состояния БД) и сравнить

сериализации. Что должны содержать такие транзакции, чтобы случилась аномалия
сериализации? Для этого каждая из транзакций работает с одними и теми же двумя
